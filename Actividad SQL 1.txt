CREATE DATABASE IF NOT EXISTS EMPRESA;
USE EMPRESA;

CREATE TABLE EMPLEADOS (
    IDEmpleado INT AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(18),
    Apellido VARCHAR(20) NOT NULL,
    Sector ENUM('Adm','Ventas','Compras','RRHH'),
    Edad INT NOT NULL DEFAULT 35,
    Sexo CHAR(1) NOT NULL CHECK (Sexo IN ('F','M')),
    Barrio_vivienda VARCHAR(30) DEFAULT 'Belgrano',
    CodPostal CHAR(4),
    Ingreso YEAR NOT NULL,
    Sueldo DECIMAL(10,2) NOT NULL,
    Antiguedad DECIMAL(10,2) NOT NULL DEFAULT 250,
    Celular VARCHAR(15) NOT NULL
);

SELECT * FROM EMPLEADOS;

SELECT Nombre, Apellido
FROM EMPLEADOS
WHERE Barrio_vivienda = 'Belgrano';

SELECT Apellido, Edad
FROM EMPLEADOS
WHERE Ingreso BETWEEN 2010 AND 2012;

SELECT COUNT(*) AS Cantidad_Mujeres
FROM EMPLEADOS
WHERE Sexo = 'F'
AND Sector = 'RRHH';

SELECT Apellido, Sector, Celular
FROM EMPLEADOS;

SELECT Nombre, Apellido, Sueldo
FROM EMPLEADOS
WHERE Barrio_vivienda IN ('Almagro','Flores')
ORDER BY Apellido DESC;

SELECT COUNT(*) AS Total_Empleados
FROM EMPLEADOS;

SELECT Nombre, Apellido, Edad
FROM EMPLEADOS
WHERE Edad = (SELECT MAX(Edad) FROM EMPLEADOS);

SELECT Nombre, Apellido, Edad
FROM EMPLEADOS
WHERE Edad = (SELECT MIN(Edad) FROM EMPLEADOS);

SELECT AVG(Edad) AS Promedio_Edades
FROM EMPLEADOS;

SELECT AVG(Edad) AS Promedio_Edad_RRHH
FROM EMPLEADOS
WHERE Sector = 'RRHH';

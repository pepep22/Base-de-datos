CREATE DATABASE IF NOT EXISTS PokemonDB;
USE PokemonDB;

CREATE TABLE IF NOT EXISTS Gimnasio (
    ID_Gimnasio INT PRIMARY KEY AUTO_INCREMENT,
    Ciudad VARCHAR(50),
    Entrenadores INT,
    BatallasRealizadas INT,
    Lider VARCHAR(50)
);

CREATE TABLE IF NOT EXISTS Pokemons (
    ID_Pokemon INT PRIMARY KEY AUTO_INCREMENT,
    TipoDeElemento VARCHAR(30),
    NivelDePoder INT,
    Batallas INT
);

CREATE TABLE IF NOT EXISTS Entrenadores (
    ID_Entrenador INT PRIMARY KEY AUTO_INCREMENT,
    Nombre VARCHAR(50),
    Edad INT,
    Gimnasio INT,
    Ciudad VARCHAR(50),
    Experiencia INT,
    Victorias INT,
    Pokemons INT,
    FOREIGN KEY (Gimnasio) REFERENCES Gimnasio(ID_Gimnasio),
    FOREIGN KEY (Pokemons) REFERENCES Pokemons(ID_Pokemon)
);

CREATE TABLE IF NOT EXISTS BatallasPokemons (
    ID_Batalla INT PRIMARY KEY AUTO_INCREMENT,
    Ganador INT,
    Perdedor INT,
    Participantes VARCHAR(100),
    Fecha DATE,
    FOREIGN KEY (Ganador) REFERENCES Pokemons(ID_Pokemon),
    FOREIGN KEY (Perdedor) REFERENCES Pokemons(ID_Pokemon)
);

SELECT e.Nombre, e.Ciudad, g.ID_Gimnasio
FROM Entrenadores e
INNER JOIN Gimnasio g
ON e.Gimnasio = g.ID_Gimnasio;

SELECT ID_Pokemon, TipoDeElemento
FROM Pokemons;

SELECT *
FROM Pokemons
WHERE NivelDePoder > 50;

SELECT p.ID_Pokemon,
       p.TipoDeElemento,
       p.NivelDePoder,
       e.Nombre AS Entrenador
FROM Pokemons p
INNER JOIN Entrenadores e
ON e.Pokemons = p.ID_Pokemon
WHERE p.TipoDeElemento = 'Fuego'
ORDER BY p.NivelDePoder ASC;

SELECT ID_Pokemon
FROM Pokemons
WHERE ID_Pokemon IN (
    SELECT Pokemons
    FROM Entrenadores
    WHERE Ciudad = 'Celadon'
);

SELECT *
FROM Pokemons
WHERE NivelDePoder > (
    SELECT AVG(NivelDePoder)
    FROM Pokemons
);

SELECT *
FROM Pokemons
WHERE ID_Pokemon IN (
    SELECT Ganador
    FROM BatallasPokemons
);

DELIMITER //

CREATE PROCEDURE SubirNivel(
    IN pEntrenador INT,
    IN pNivel INT
)
BEGIN
    UPDATE Pokemons
    SET NivelDePoder = NivelDePoder + pNivel
    WHERE ID_Pokemon = (
        SELECT Pokemons
        FROM Entrenadores
        WHERE ID_Entrenador = pEntrenador
    );
END //

DELIMITER ;

CALL SubirNivel(1,5);

DELIMITER //

CREATE PROCEDURE RegistrarBatalla(
    IN pGanador INT,
    IN pPerdedor INT,
    IN pParticipantes VARCHAR(100),
    IN pFecha DATE
)
BEGIN
    INSERT INTO BatallasPokemons
    (Ganador, Perdedor, Participantes, Fecha)
    VALUES
    (pGanador, pPerdedor, pParticipantes, pFecha);
END //

DELIMITER ;

CALL RegistrarBatalla(1,2,'Pikachu vs Charizard','2026-06-27');

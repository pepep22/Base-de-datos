CREATE DATABASE IF NOT EXISTS series;
USE series;

CREATE TABLE series (
IDseries INT AUTO_INCREMENT PRIMARY KEY,
nombre VARCHAR (30),
genero VARCHAR (15),
añolanzamiento VARCHAR (14),
presupuesto INT (10),
plataforma VARCHAR (15),
director VARCHAR (25),
edaddirector INT (3),
sueldodirector INT (10)
);

Create Table temporadas (
IDtemporadas INT auto_increment Primary key,
NombreTemporada VARCHAR (20),
DescripcionTemporada VARCHAR (30),
IDseries INT,
FOREIGN KEY (IDseries) REFERENCES series(IDseries)
);

CREATE TABLE Personajes (
IDpersonajes INT AUTO_INCREMENT PRIMARY KEY,
NombrePersonaje VARCHAR (30),
Edad VARCHAR (5),
Genero VARCHAR (10),
IDTemporadas INT,
FOREIGN KEY (IDTemporadas) REFERENCES temporadas(IDtemporadas),
ID_Series INT,
FOREIGN KEY (ID_Series) REFERENCES series(IDseries)
);

CREATE TABLE Episodios (
IDepisodio INT AUTO_INCREMENT PRIMARY KEY,
NombreEpisodio VARCHAR (30), 
Descripcion VARCHAR (300),
Puntuacion FLOAT (6),
fechaestreno INT (14),
IDTemporadas INT,
FOREIGN KEY (IDTemporadas) REFERENCES temporadas(IDtemporadas),
ID_Series INT,
FOREIGN KEY (ID_Series) REFERENCES series(IDseries)
);

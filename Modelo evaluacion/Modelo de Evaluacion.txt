CREATE DATABASE IF NOT EXISTS Tienda;
USE Tienda;
CREATE TABLE Productos (
    ID_Productos INT AUTO_INCREMENT PRIMARY KEY,
    NombreProducto VARCHAR(100) NOT NULL,
    Descripcion VARCHAR(255),
    Precio DECIMAL(10,2) NOT NULL,
    StockDisponible INT NOT NULL
);
CREATE TABLE Clientes (
    ID_Clientes INT AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    Apellido VARCHAR(50) NOT NULL,
    CorreoElectronico VARCHAR(100) UNIQUE,
    Direccion VARCHAR(150)
);
CREATE TABLE Pedidos(
    ID_Pedidos INT AUTO_INCREMENT PRIMARY KEY,
    ID_Productos INT,
    ID_Clientes INT,
    FechaPedido DATE,
    EstadoPedido VARCHAR(30),
    PrecioTotal DECIMAL(10,2),
    TotalProductos INT,
    FOREIGN KEY (ID_Productos)
        REFERENCES Productos(ID_Productos),
    FOREIGN KEY (ID_Clientes)
        REFERENCES Clientes(ID_Clientes)
);

SELECT P.NombreProducto, P.Precio
FROM Productos P
WHERE P.Precio > 50;


SELECT DISTINCT C.Nombre, C.Apellido
FROM Clientes C
INNER JOIN Pedidos P
ON C.ID_Clientes = P.ID_Clientes;


SELECT C.Nombre, C.Apellido, P.FechaPedido
FROM Clientes C
INNER JOIN Pedidos P
ON C.ID_Clientes = P.ID_Clientes
WHERE P.EstadoPedido = 'Entregado';


SELECT P.EstadoPedido, C.CorreoElectronico
FROM Pedidos P
INNER JOIN Clientes C
ON P.ID_Clientes = C.ID_Clientes;


SELECT C.Nombre, C.Apellido, P.FechaPedido
FROM Clientes C
INNER JOIN Pedidos P
ON C.ID_Clientes = P.ID_Clientes
WHERE P.FechaPedido = (
    SELECT MAX(FechaPedido)
    FROM Pedidos
);


SELECT C.Nombre, C.Apellido, P.FechaPedido, P.EstadoPedido
FROM Clientes C
INNER JOIN Pedidos P
ON C.ID_Clientes = P.ID_Clientes
WHERE P.EstadoPedido = 'Pendiente';


SELECT NombreProducto, Precio
FROM Productos
WHERE Precio = (
    SELECT MAX(Precio)
    FROM Productos

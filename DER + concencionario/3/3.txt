CREATE DATABASE IF NOT EXISTS LigaFutbol;
USE LigaFutbol;

CREATE TABLE Equipos (
    id_equipo INT PRIMARY KEY,
    nombre_equipo VARCHAR(100),
    nombre_estadio VARCHAR(100),
    aforo INT,
    anio_fundacion INT,
    ciudad VARCHAR(100)
);

CREATE TABLE Jugadores (
    id_jugador INT PRIMARY KEY,
    nombre VARCHAR(100),
    fecha_nacimiento DATE,
    posicion VARCHAR(50),
    id_equipo INT,
    FOREIGN KEY (id_equipo) REFERENCES Equipos(id_equipo)
);

CREATE TABLE Presidentes (
    id_presidente INT PRIMARY KEY,
    nombre VARCHAR(100),
    apellido VARCHAR(100),
    dni VARCHAR(15) UNIQUE,
    fecha_nacimiento DATE,
    id_equipo INT UNIQUE,
    anio_presidente_elegido INT,
    FOREIGN KEY (id_equipo) REFERENCES Equipos(id_equipo)
);

CREATE TABLE Partidos (
    id_partido INT PRIMARY KEY,
    fecha DATE,
    goles_equipo_casa INT,
    goles_equipo_fuera INT
);

CREATE TABLE Goles (
    id_gol INT PRIMARY KEY,
    id_partido INT,
    id_jugador INT,
    minuto_del_gol INT,
    descripcion_del_gol VARCHAR(255),
    FOREIGN KEY (id_partido) REFERENCES Partidos(id_partido),
    FOREIGN KEY (id_jugador) REFERENCES Jugadores(id_jugador)
);

SELECT j.nombre, e.nombre_equipo
FROM Jugadores j
JOIN Equipos e ON j.id_equipo = e.id_equipo;

SELECT p.nombre, p.apellido, e.nombre_equipo
FROM Presidentes p
JOIN Equipos e ON p.id_equipo = e.id_equipo;

SELECT g.id_gol, j.nombre AS jugador, g.minuto_del_gol, g.descripcion_del_gol
FROM Goles g
JOIN Jugadores j ON g.id_jugador = j.id_jugador;

FROM Equipos;

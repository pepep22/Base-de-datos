CREATE DATABASE IF NOT EXISTS Concesionario;
USE Concesionario;

CREATE TABLE Clientes (
    id_cliente INT PRIMARY KEY,
    dni VARCHAR(15) UNIQUE,
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    direccion VARCHAR(100),
    telefono VARCHAR(20)
);

CREATE TABLE Coches (
    id_coche INT PRIMARY KEY,
    matricula VARCHAR(15) UNIQUE,
    modelo VARCHAR(50),
    marca VARCHAR(50),
    color VARCHAR(30),
    id_cliente INT,
    FOREIGN KEY (id_cliente) REFERENCES Clientes(id_cliente)
);

CREATE TABLE CochesNuevos (
    id_coche_nuevo INT PRIMARY KEY,
    nro_unidades INT,
    concesionario VARCHAR(100),
    id_coche INT,
    FOREIGN KEY (id_coche) REFERENCES Coches(id_coche)
);

CREATE TABLE CochesViejos (
    id_coche_viejo INT PRIMARY KEY,
    kilometros_recorridos INT,
    id_coche INT,
    FOREIGN KEY (id_coche) REFERENCES Coches(id_coche)
);

CREATE TABLE Mecanicos (
    id_mecanico INT PRIMARY KEY,
    dni VARCHAR(15) UNIQUE,
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    fecha_contratacion DATE,
    salario DECIMAL(10,2)
);

CREATE TABLE Reparacion (
    id_reparacion INT PRIMARY KEY,
    id_coche INT,
    id_mecanico INT,
    fecha_reparacion DATE,
    tiempo INT,
    FOREIGN KEY (id_coche) REFERENCES Coches(id_coche),
    FOREIGN KEY (id_mecanico) REFERENCES Mecanicos(id_mecanico)
);

SELECT c.nombre, c.apellido, co.matricula, co.modelo, co.marca
FROM Clientes c
JOIN Coches co ON c.id_cliente = co.id_cliente;

SELECT co.matricula, co.modelo, r.fecha_reparacion, r.tiempo
FROM Coches co
JOIN Reparacion r ON co.id_coche = r.id_coche;

SELECT m.nombre, m.apellido, co.matricula, r.fecha_reparacion
FROM Mecanicos m
JOIN Reparacion r ON m.id_mecanico = r.id_mecanico
JOIN Coches co ON r.id_coche = co.id_coche;

FROM Clientes;

FROM Coches;

CREATE DATABASE IF NOT EXISTS CampeonatoFutbol;
USE CampeonatoFutbol;

CREATE TABLE Equipo (
    ID_Equipo INT PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Jugador (
    ID_Jugador INT PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    Posicion VARCHAR(30) NOT NULL,
    ID_Equipo INT,
    FOREIGN KEY (ID_Equipo) REFERENCES Equipo(ID_Equipo)
);

CREATE TABLE Estadio (
    ID_Estadio INT PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL,
    Ciudad VARCHAR(50)
);

CREATE TABLE Partido (
    ID_Partido INT PRIMARY KEY,
    Fecha DATE NOT NULL,
    ID_Equipo_Local INT,
    ID_Equipo_Visitante INT,
    Goles_Local INT,
    Goles_Visitante INT,
    Resultado VARCHAR(20),
    ID_Estadio INT,
    FOREIGN KEY (ID_Equipo_Local) REFERENCES Equipo(ID_Equipo),
    FOREIGN KEY (ID_Equipo_Visitante) REFERENCES Equipo(ID_Equipo),
    FOREIGN KEY (ID_Estadio) REFERENCES Estadio(ID_Estadio)
);

SELECT
    p.ID_Partido,
    p.Fecha,
    el.Nombre AS Equipo_Local,
    ev.Nombre AS Equipo_Visitante,
    p.Goles_Local,
    p.Goles_Visitante,
    p.Resultado,
    e.Nombre AS Estadio
FROM Partido p
INNER JOIN Equipo el
    ON p.ID_Equipo_Local = el.ID_Equipo
INNER JOIN Equipo ev
    ON p.ID_Equipo_Visitante = ev.ID_Equipo
INNER JOIN Estadio e
    ON p.ID_Estadio = e.ID_Estadio;

SELECT
    Nombre AS Jugador,
    Posicion
FROM Jugador;
